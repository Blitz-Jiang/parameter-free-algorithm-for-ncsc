Suppose the algorithm completes at least $K$ accepted steps, where $K\ge1$ is finite.
If $K\le j_0$, then $K=O(\mathcal L)$ by \eqref{eq:initial-count}.
Otherwise, set
\[
 m=K-j_0,\qquad \mathcal J=\{j_0,\ldots,K-1\}.
\]
\begin{equation}\label{eq:summation-bounds}
 g_{j_0}\underset{\eqref{eq:initial-gradient}}{\le}C(\Delta^2Q)^{1/3},\qquad
 \sum_{j\in\mathcal J}\sigma_{\tau_j}
       \underset{\eqref{eq:scale-sum}}{\le}C(mL_0+L_1^{3/2}\Delta),\qquad
 \sum_{j\in\mathcal J}\sigma_{\tau_j}r_{\tau_j}^3
       \underset{\eqref{eq:descent-sum}}{\le}\Delta/3.
\end{equation}
For $p=1,2$, H\"older's inequality gives
\begin{align}
 \sum_{j\in\mathcal J\cap\mathcal B}\sigma_{\tau_j}r_{\tau_j}^p
 &\le
       \left(\sum_{j\in\mathcal J\cap\mathcal B}
                         \sigma_{\tau_j}r_{\tau_j}^3\right)^{p/3}
       \left(\sum_{j\in\mathcal J\cap\mathcal B}
                         \sigma_{\tau_j}\right)^{1-p/3}\nonumber\\
 &\underset{\eqref{eq:summation-bounds}}{\le}C\Delta^{p/3}
             (mL_0+L_1^{3/2}\Delta)^{1-p/3}.
 \label{eq:boundary-sums}
\end{align}

Summing the boundary and interior estimates and using
$\sum_{j\in\mathcal J}(g_{j+1}-g_j)=g_K-g_{j_0}$ gives
\begin{align}
 \tfrac5{16}\sum_{j\in\mathcal J}g_j
 &\underset{\eqref{eq:boundary},\,\eqref{eq:interior}}{\le}g_{j_0}-g_K
       +51\sum_{j\in\mathcal J\cap\mathcal B}
                        \sigma_{\tau_j}r_{\tau_j}^2\nonumber\\
 &\underset{\eqref{eq:summation-bounds},\,\eqref{eq:boundary-sums}}{\le}C\Delta^{2/3}Q^{1/3}
       +C\Delta^{2/3}(mL_0+L_1^{3/2}\Delta)^{1/3}\nonumber\\
 &\le C\Delta^{2/3}
                         (mL_0+L_1^{3/2}\Delta)^{1/3}.
 \label{eq:sum-g}
\end{align}
Here $Q\le mL_0+L_1^{3/2}\Delta$ since $m\ge1$.

For the curvature sum,
\begin{align}
 \sum_{j\in\mathcal J}\chi_j
 &\underset{\eqref{eq:boundary},\,\eqref{eq:interior}}{\le}
       32\sum_{j\in\mathcal J\cap\mathcal B}
                         \sigma_{\tau_j}r_{\tau_j}
       +2\sum_{j\in\mathcal J\cap\mathcal I}
                         \sqrt{\sigma_{\tau_j}g_j}\nonumber\\
 &\le
       32\sum_{j\in\mathcal J\cap\mathcal B}
                         \sigma_{\tau_j}r_{\tau_j}
       +2\left(\sum_{j\in\mathcal J}\sigma_{\tau_j}\right)^{1/2}
          \left(\sum_{j\in\mathcal J}g_j\right)^{1/2}\nonumber\\
 &\underset{\eqref{eq:summation-bounds},\,\eqref{eq:boundary-sums},\,\eqref{eq:sum-g}}{\le}C\Delta^{1/3}
                         (mL_0+L_1^{3/2}\Delta)^{2/3}.
 \label{eq:sum-curvature}
\end{align}

Since the stopping test fails for every $j\in\mathcal J$,
\begin{align}
 m
 &<\frac1\epsilon\sum_{j\in\mathcal J}g_j
       +\frac1\delta\sum_{j\in\mathcal J}\chi_j\nonumber\\
 &\underset{\eqref{eq:sum-g},\,\eqref{eq:sum-curvature}}{\le}C\left[
 \frac{\Delta^{2/3}L_0^{1/3}}{\epsilon}m^{1/3}
 +\frac{\sqrt{L_1}\Delta}{\epsilon}
 +\frac{\Delta^{1/3}L_0^{2/3}}{\delta}m^{2/3}
 +\frac{L_1\Delta}{\delta}\right]\nonumber\\
 &\le\frac m2+C\Delta\left[
 \frac{\sqrt{L_0}}{\epsilon^{3/2}}
 +\frac{\sqrt{L_1}}{\epsilon}
 +\frac{L_0^2}{\delta^3}
 +\frac{L_1}{\delta}\right].
 \label{eq:accepted-absorption}
\end{align}
The last inequality follows from Young's inequality in the forms
\[
 \begin{aligned}
 C\frac{\Delta^{2/3}L_0^{1/3}}{\epsilon}m^{1/3}
 &\le\frac m4+C\Delta\frac{\sqrt{L_0}}{\epsilon^{3/2}},\\
 C\frac{\Delta^{1/3}L_0^{2/3}}{\delta}m^{2/3}
 &\le\frac m4+C\Delta\frac{L_0^2}{\delta^3}.
 \end{aligned}
\]
Subtracting $m/2$ and using $K=j_0+m$ proves
\begin{equation}\label{eq:accepted-count}
 K\le j_0+C\Delta\left[
 \frac{\sqrt{L_0}}{\epsilon^{3/2}}
 +\frac{\sqrt{L_1}}{\epsilon}
 +\frac{L_0^2}{\delta^3}
 +\frac{L_1}{\delta}\right].
\end{equation}
Thus \eqref{eq:accepted-count} holds whenever the algorithm completes at least
$K$ accepted steps. By Lemma~\ref{lem:safe}, an infinite run would complete
arbitrarily many accepted steps, contradicting this uniform bound.
The algorithm therefore terminates at an
$(\epsilon,\delta)$-second-order stationary point.

Initial termination costs at most three oracle evaluations.
Otherwise, let $K\ge1$ and $U\ge0$ count the accepted and rejected
trials before termination. Since a rejection leaves the point
unchanged, termination follows an accepted step. Hence
\[
 \sigma_{\tau_{K-1}}=\sigma_0\,2^{U-K+1},
 \qquad
 U=K-1+\log_2\frac{\sigma_{\tau_{K-1}}}{\sigma_0}.
\]
Hence,
\begin{equation}\label{eq:rejected-count}
 U\underset{\eqref{eq:potential-expansion}}{\le}
       K+C\left(1+\lp\frac{Q_0}{\sigma_0}\right).
\end{equation}
There are $K+U$ subproblem solves. Reusing the current values and
derivatives as specified in Algorithm~\ref{alg:utr} gives
\[
 N_f\le1+K+U,\qquad
 N_{\nabla f}\le1+K+U,\qquad
 N_{\nabla^2 f}\le1+K.
\]
It follows that
\begin{align}
 \max\{K+U,N_f+N_{\nabla f}+N_{\nabla^2f}\}
 &\le3+3K+2U\nonumber\\
 &\underset{\eqref{eq:rejected-count}}{\le}5K+C\left(1+\lp\frac{Q_0}{\sigma_0}\right)\nonumber\\
 &\underset{\eqref{eq:accepted-count},\,\eqref{eq:initial-count},\,\eqref{eq:logs}}{\le}C\Delta\left[
 \frac{\sqrt{L_0}}{\epsilon^{3/2}}
 +\frac{\sqrt{L_1}}{\epsilon}
 +\frac{L_0^2}{\delta^3}
 +\frac{L_1}{\delta}\right]+C\mathcal L.
 \label{eq:total-count}
\end{align}
Since $O(\mathcal L)=\widetilde O(1)$ is independent of
$\epsilon,\delta$, \eqref{eq:total-count} proves
\eqref{eq:complexity}. The algorithm uses neither $L_0$ nor $L_1$,
so the bound holds for every admissible pair along the same run.
