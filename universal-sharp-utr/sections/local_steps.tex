\subsection{Nonzero steps and finite backtracking}

Write $r_k=\norm{s_k}$.

\begin{lemma}\label{lem:safe}
A trial is accepted whenever
\begin{equation}\label{eq:safe}
 \sigma_k\ge L_0+L_1\norm{\nabla f(x_k)}
        +\sqrt{L_1}[-\lambda_{\min}(\nabla^2 f(x_k))]_+.
\end{equation}
Consequently, backtracking terminates after finitely many rejections.
\end{lemma}
\begin{proof}
We first show that $r_k>0$. If $s_k=0$, then \eqref{eq:kkt-stationarity}
gives $\nabla f(x_k)=0$, while the failed stopping test and \eqref{eq:model-scales}
imply that the model Hessian has a negative eigenvalue. Since $R_k>0$,
this contradicts the global optimality of $s_k=0$.

Now fix a trial satisfying \eqref{eq:safe}.
Let $M_k=\max_{0\le u\le1}\norm{\nabla^3 f(x_k+us_k)}$ as in
Lemma~\ref{lem:taylor-remainders}. For the function decrease,
\begin{subequations}
\begin{equation}\label{eq:trial-value-remainder}
\begin{aligned}
 f(x_k)-f(x_k+s_k)
 &\underset{\eqref{eq:taylor-value}}{\ge}
 -\nabla f(x_k)^Ts_k-\tfrac12s_k^T\nabla^2 f(x_k)s_k-\tfrac16M_kr_k^3\\
 &\underset{\eqref{eq:kkt-stationarity}}{=}
 \tfrac12s_k^T\nabla^2 f(x_k)s_k
 +(2\sqrt{\sigma_kh_k}+\lambda_k)r_k^2-\tfrac16M_kr_k^3\\
 &\underset{\eqref{eq:kkt-psd}}{\ge}
 \bigl(\sqrt{\sigma_kh_k}+\tfrac12\lambda_k\bigr)r_k^2-\tfrac16M_kr_k^3.
\end{aligned}
\end{equation}
For the gradient at the trial point,
\begin{equation}\label{eq:trial-gradient-remainder}
\begin{aligned}
 \norm{\nabla f(x_k+s_k)}
 &\underset{\eqref{eq:taylor-gradient}}{\le}
 \norm{\nabla f(x_k)+\nabla^2 f(x_k)s_k}+\tfrac12M_kr_k^2\\
 &\underset{\eqref{eq:kkt-stationarity}}{=}
 (2\sqrt{\sigma_kh_k}+\lambda_k)r_k+\tfrac12M_kr_k^2.
\end{aligned}
\end{equation}
\end{subequations}

To bound $M_k$ using \eqref{eq:assumption}, we use the following estimate,
which holds for every trial. Taking norms in \eqref{eq:kkt-stationarity}
gives the first equality below.
Taking the inner product of the same condition with $s_k/r_k$ gives the second.
We then use Cauchy--Schwarz and the Rayleigh quotient bound
$s_k^T\nabla^2f(x_k)s_k\ge\lambda_{\min}(\nabla^2 f(x_k))r_k^2$ to obtain
\begin{equation}\label{eq:linear-gradient}
 \begin{aligned}
 \norm{\nabla f(x_k)+\nabla^2f(x_k)s_k}
 &= (2\sqrt{\sigma_kh_k}+\lambda_k)r_k\\
 &= -\frac{\nabla f(x_k)^Ts_k}{r_k}
    -\frac{s_k^T\nabla^2f(x_k)s_k}{r_k}\\
 &\le \norm{\nabla f(x_k)}+[-\lambda_{\min}(\nabla^2f(x_k))]_+r_k.
 \end{aligned}
\end{equation}
For $0\le u\le1$, the vector $\nabla f(x_k)+u\nabla^2f(x_k)s_k$
is the convex combination of $\nabla f(x_k)$ and
$\nabla f(x_k)+\nabla^2f(x_k)s_k$ with weights $1-u$ and $u$.
By convexity of the norm,
\begin{equation}\label{eq:linear-segment}
 \begin{aligned}
 \norm{\nabla f(x_k)+u\nabla^2f(x_k)s_k}
 &\le (1-u)\norm{\nabla f(x_k)}
       +u\norm{\nabla f(x_k)+\nabla^2f(x_k)s_k}\\
 &\underset{\eqref{eq:linear-gradient}}{\le}
       \norm{\nabla f(x_k)}+u[-\lambda_{\min}(\nabla^2f(x_k))]_+r_k\\
 &\le \norm{\nabla f(x_k)}+[-\lambda_{\min}(\nabla^2f(x_k))]_+r_k.
 \end{aligned}
\end{equation}

By \eqref{eq:model-scales} and \eqref{eq:safe},
\[
 L_1h_k
 =\max\left\{L_1\norm{\nabla f(x_k)},
       \frac{L_1[-\lambda_{\min}(\nabla^2f(x_k))]_+^2}
            {64\sigma_k}\right\}
 \le\sigma_k.
\]
Hence,
\begin{equation}\label{eq:safe-radius}
 \sqrt{L_1}r_k
 \le\sqrt{L_1}R_k=
       \frac14\sqrt{\frac{L_1h_k}{\sigma_k}}
 \le\frac14,\qquad L_1r_k^2\le\frac1{16}.
\end{equation}
For $0\le u\le1$, apply \eqref{eq:taylor-gradient} to the step $us_k$.
The maximum of $\norm{\nabla^3f}$ on this shorter segment is at most $M_k$, so
\begin{equation}\label{eq:segment-gradient}
 \begin{aligned}
 \norm{\nabla f(x_k+us_k)}
 &\le\norm{\nabla f(x_k)+u\nabla^2f(x_k)s_k}
       +\tfrac12M_kr_k^2\\
 &\underset{\eqref{eq:linear-segment}}{\le}
       \norm{\nabla f(x_k)}
       +[-\lambda_{\min}(\nabla^2f(x_k))]_+r_k
       +\tfrac12M_kr_k^2.
 \end{aligned}
\end{equation}
Taking maxima along the segment gives
\[
 \begin{aligned}
 M_k
 &\underset{\eqref{eq:assumption},\,\eqref{eq:segment-gradient}}{\le}
       L_0+L_1\norm{\nabla f(x_k)}
       +L_1[-\lambda_{\min}(\nabla^2f(x_k))]_+r_k
       +\tfrac12L_1M_kr_k^2\\
 &\underset{\eqref{eq:safe-radius}}{\le}
       L_0+L_1\norm{\nabla f(x_k)}
       +\tfrac14\sqrt{L_1}[-\lambda_{\min}(\nabla^2f(x_k))]_+
       +\tfrac1{32}M_k\\
 &\underset{\eqref{eq:safe}}{\le}\sigma_k+\tfrac1{32}M_k.
 \end{aligned}
\]
Rearranging gives
\begin{equation}\label{eq:third-derivative-bound}
 M_k\le\frac{32}{31}\sigma_k.
\end{equation}
By \eqref{eq:third-derivative-bound}, $M_k/6\le\sigma_k$ and
$M_k/2\le3\sigma_k$. Substituting in
\eqref{eq:trial-value-remainder} and \eqref{eq:trial-gradient-remainder}
and using $r_k\le R_k=\sqrt{h_k}/(4\sqrt{\sigma_k})$ gives
\begin{align*}
 f(x_k)-f(x_k+s_k)
 &\ge \bigl(\sqrt{\sigma_kh_k}+\tfrac12\lambda_k-\sigma_kr_k\bigr)r_k^2
 \ge \bigl(\tfrac34\sqrt{\sigma_kh_k}+\tfrac12\lambda_k\bigr)r_k^2,\\
 \norm{\nabla f(x_k+s_k)}
 &\le (2\sqrt{\sigma_kh_k}+\lambda_k)r_k+3\sigma_kr_k^2\\
 &\le \tfrac12h_k+\lambda_kr_k+\tfrac3{16}h_k
 =\tfrac{11}{16}h_k+\lambda_kr_k.
\end{align*}
Thus \eqref{eq:accept-value} and \eqref{eq:accept-gradient} hold.
Rejections leave the iterate, and hence the finite threshold in
\eqref{eq:safe}, unchanged while doubling the scale, so
backtracking terminates after finitely many doublings.
\end{proof}

\subsection{Boundary and interior steps}
\label{sec:accepted-steps}
Let $\tau_j$ denote the trial index of the $(j+1)$st accepted step, starting
with $j=0$. Write $y_j$ for the iterate after $j$ accepted steps, so that
\begin{equation}\label{eq:accepted-points}
 y_0=x_0,\qquad y_{j+1}=x_{\tau_j}+s_{\tau_j}.
\end{equation}
Since rejected trials leave the iterate unchanged, $x_{\tau_j}=y_j$ at
each accepted step. In this section and Section~\ref{sec:potential-proof},
estimates for iterates, steps, and blocks of steps apply whenever the
corresponding iterates or steps have been generated.
Define
\begin{equation}\label{eq:accepted-measures}
 g_j=\norm{\nabla f(y_j)},\qquad
 \chi_j=[-\lambda_{\min}(\nabla^2f(y_j))]_+,\qquad
 \Delta_j=f(y_j)-f(y_{j+1}).
\end{equation}
Partition the accepted steps into boundary and interior steps:
\begin{equation}\label{eq:step-sets}
 \mathcal B=\{j:r_{\tau_j}=R_{\tau_j}\},\qquad
 \mathcal I=\{j:r_{\tau_j}<R_{\tau_j}\}.
\end{equation}

\begin{lemma}\label{lem:geometry}
For every $j\in\mathcal B$,
\begin{equation}\label{eq:boundary}
 \begin{aligned}
 g_j&\le16\sigma_{\tau_j}r_{\tau_j}^2,&
 g_{j+1}&\le51\sigma_{\tau_j}r_{\tau_j}^2,\\
 \chi_j&\le32\sigma_{\tau_j}r_{\tau_j},&
 \lambda_{\tau_j}&\le10\sqrt{\sigma_{\tau_j}h_{\tau_j}}.
 \end{aligned}
\end{equation}
For every $j\in\mathcal I$,
\begin{equation}\label{eq:interior}
 h_{\tau_j}=g_j,\qquad \lambda_{\tau_j}=0,\qquad
 g_{j+1}\le\tfrac{11}{16}g_j,\qquad
 \chi_j\le2\sqrt{\sigma_{\tau_j}g_j}.
\end{equation}
\end{lemma}
\begin{proof}
By \eqref{eq:model-scales} and \eqref{eq:accepted-measures},
$g_j\le h_{\tau_j}$ and
$\chi_j\le8\sqrt{\sigma_{\tau_j}h_{\tau_j}}$.
If $j\in\mathcal B$, then $r_{\tau_j}=R_{\tau_j}$, so
\begin{equation}\label{eq:boundary-identities}
 h_{\tau_j}\underset{\eqref{eq:model-scales}}{=}16\sigma_{\tau_j}r_{\tau_j}^2,
 \qquad \sqrt{\sigma_{\tau_j}h_{\tau_j}}
       =4\sigma_{\tau_j}r_{\tau_j}.
\end{equation}
Since $r_{\tau_j}=R_{\tau_j}>0$,
\begin{equation}\label{eq:boundary-multiplier}
 \begin{aligned}
 \lambda_{\tau_j}
 &\underset{\eqref{eq:linear-gradient}}{\le}
       \frac{g_j}{R_{\tau_j}}+\chi_j
       -2\sqrt{\sigma_{\tau_j}h_{\tau_j}}\\
 &\le
       (4+8-2)\sqrt{\sigma_{\tau_j}h_{\tau_j}}
       =10\sqrt{\sigma_{\tau_j}h_{\tau_j}}.
 \end{aligned}
\end{equation}
The gradient test gives
\[
 \begin{aligned}
 g_{j+1}&\underset{\eqref{eq:accept-gradient},\,\eqref{eq:boundary-multiplier}}{\le}
       \tfrac{11}{16}h_{\tau_j}
       +10\sqrt{\sigma_{\tau_j}h_{\tau_j}}\,r_{\tau_j}\\
 &\underset{\eqref{eq:boundary-identities}}{=}
       (11+40)\sigma_{\tau_j}r_{\tau_j}^2
       =51\sigma_{\tau_j}r_{\tau_j}^2.
 \end{aligned}
\]

If $j\in\mathcal I$, complementary slackness
\eqref{eq:kkt-complementarity} gives $\lambda_{\tau_j}=0$, and
\eqref{eq:kkt-psd} yields $\chi_j\le2\sqrt{\sigma_{\tau_j}h_{\tau_j}}$.
If $h_{\tau_j}>g_j$, then \eqref{eq:model-scales} implies $\chi_j>0$,
and hence
\[
 \chi_j\le
 2\sqrt{\sigma_{\tau_j}h_{\tau_j}}
 =\frac{\chi_j}{4}<\chi_j,
\]
a contradiction. Thus $h_{\tau_j}=g_j$.
The curvature bound above and \eqref{eq:accept-gradient} give
the remaining estimates.
\end{proof}
