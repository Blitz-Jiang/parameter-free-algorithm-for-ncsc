\subsection{Global convergence and universal sharp complexity}
\label{sec:main-theorem}

\begin{theorem}[Universal sharp complexity]\label{thm:main}
Under Assumption~\ref{ass:setting}, Algorithm~\ref{alg:utr} finds an
$(\epsilon,\delta)$-second-order stationary point within
\begin{equation}\label{eq:complexity}
 O\!\left(\Delta\left[
 \frac{\sqrt{L_0}}{\epsilon^{3/2}}+
 \frac{\sqrt{L_1}}{\epsilon}+
 \frac{L_0^2}{\delta^3}+
 \frac{L_1}{\delta}\right]\right)+\widetilde O(1)
\end{equation}
trust-region subproblem solves and the same order of total exact
function, gradient, and Hessian evaluations.
\end{theorem}

If $\Delta=0$, then $x_0$ is a global minimizer and satisfies the stopping
test. Hence assume $\Delta>0$.
