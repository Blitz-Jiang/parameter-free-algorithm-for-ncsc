\subsection{Scale control and initial gradient reduction}
\label{sec:potential-proof}

\begin{lemma}[One-step scale and gradient estimates]\label{lem:young}
For $0\le j<K$ and every $\eta>0$,
\begin{align}
 \sqrt{L_1}\chi_j
 &\le\frac{\sigma_{\tau_j}}{16}+256L_1g_j
                         +CL_1^{3/2}\Delta_j,\label{eq:curvature-young}\\
 L_1g_{j+1}
 &\le\tfrac{11}{16}L_1g_j+\eta\sigma_{\tau_j}
                         +\frac{C}{\sqrt\eta}L_1^{3/2}\Delta_j.
                         \label{eq:gradient-young}
\end{align}
Consequently,
\begin{equation}\label{eq:scale-recursion}
 \sigma_{\tau_j}\le\tfrac47\sigma_{\tau_{j-1}}
                         +3L_0+600L_1g_j+CL_1^{3/2}\Delta_j.
\end{equation}
\end{lemma}
\begin{proof}
If $h_{\tau_j}=g_j$, then
\[
 \sqrt{L_1}\chi_j
       \underset{(a)}{\le}8\sqrt{\sigma_{\tau_j}L_1g_j}
       \underset{(b)}{\le}\frac{\sigma_{\tau_j}}{16}+256L_1g_j.
\]
Here (a) uses \eqref{eq:model-scales} and (b) is Young's inequality.
If $h_{\tau_j}>g_j$, then
$2\sqrt{\sigma_{\tau_j}h_{\tau_j}}=\chi_j/4$.
Equation~\eqref{eq:kkt-psd} implies
$\lambda_{\tau_j}\ge3\chi_j/4>0$, so
\eqref{eq:kkt-complementarity} gives $j\in\mathcal B$.
Using \eqref{eq:model-scales} and \eqref{eq:descent-sum},
\[
 r_{\tau_j}=\frac{\chi_j}{32\sigma_{\tau_j}},\qquad
 \Delta_j\ge3\sigma_{\tau_j}r_{\tau_j}^3
             =\frac{3\chi_j^3}{32^3\sigma_{\tau_j}^2}.
\]
Therefore
\[
 \sqrt{L_1}\chi_j
 \underset{(a)}{\le}\frac{32}{3^{1/3}}\sigma_{\tau_j}^{2/3}
                      (L_1^{3/2}\Delta_j)^{1/3}
 \underset{(b)}{\le}\frac{\sigma_{\tau_j}}{16}+CL_1^{3/2}\Delta_j.
\]
Here (a) rearranges the preceding bound on $\Delta_j$, and
(b) is Young's inequality with exponents $3/2$ and $3$.
These two cases prove \eqref{eq:curvature-young}.

For $j\in\mathcal I$, \eqref{eq:gradient-young} follows directly
from \eqref{eq:interior}. For $j\in\mathcal B$,
\[
 \begin{aligned}
 L_1g_{j+1}
 &\underset{(a)}{\le}51L_1\sigma_{\tau_j}r_{\tau_j}^2\\
 &\underset{(b)}{\le}\eta\sigma_{\tau_j}
          +\frac{C}{\sqrt\eta}L_1^{3/2}\sigma_{\tau_j}r_{\tau_j}^3\\
 &\underset{(c)}{\le}\eta\sigma_{\tau_j}
          +\frac{C}{\sqrt\eta}L_1^{3/2}\Delta_j.
 \end{aligned}
\]
Here (a) is \eqref{eq:boundary}, (b) is Young's inequality with
exponents $3/2$ and $3$, and (c) uses \eqref{eq:descent-sum}.
This proves \eqref{eq:gradient-young}.
Finally, \eqref{eq:backtrack-upper} and
\eqref{eq:curvature-young} imply
\[
 \sigma_{\tau_j}
 \le\tfrac12\sigma_{\tau_{j-1}}+2L_0+514L_1g_j
                  +\tfrac18\sigma_{\tau_j}+CL_1^{3/2}\Delta_j.
\]
Moving $\sigma_{\tau_j}/8$ to the left and using
$16/7<3$ and $4112/7<600$ proves \eqref{eq:scale-recursion}.
\end{proof}

\begin{proposition}[Joint potential and scale sums]\label{prop:potential}
Define
\begin{equation}\label{eq:potential-definition}
 P_j=\sigma_{\tau_{j-1}}+10^4L_1g_j\quad(0\le j\le K).
\end{equation}
Then
\begin{equation}\label{eq:potential}
 P_{j+1}\le\tfrac34P_j+4L_0+CL_1^{3/2}\Delta_j
                  \quad(0\le j<K).
\end{equation}
In particular,
\begin{equation}\label{eq:potential-expansion}
 P_j\le(3/4)^jP_0+16L_0+CL_1^{3/2}\Delta
                  \quad(0\le j\le K),
 \qquad \sigma_{\tau_j}\le CQ_0\quad(0\le j<K).
\end{equation}
For $0\le b<b+m\le K$,
\begin{equation}\label{eq:scale-sum-general}
 \sum_{j=b}^{b+m-1}\sigma_{\tau_j}
 \le3P_b+16mL_0+CL_1^{3/2}\Delta.
\end{equation}
\end{proposition}
\begin{proof}
With $\eta=10^{-6}$,
\[
 \begin{aligned}
 P_{j+1}
 &\underset{(a)}{\le}
       1.01\sigma_{\tau_j}+6875L_1g_j+CL_1^{3/2}\Delta_j\\
 &\underset{(b)}{\le}\tfrac{101}{175}\sigma_{\tau_{j-1}}
          +3.03L_0+7481L_1g_j+CL_1^{3/2}\Delta_j\\
 &\underset{(c)}{\le}\tfrac34P_j+4L_0+CL_1^{3/2}\Delta_j.
 \end{aligned}
\]
Here (a) uses \eqref{eq:gradient-young} and
\eqref{eq:potential-definition}; (b) uses \eqref{eq:scale-recursion};
and (c) uses $101/175<3/4$, $3.03<4$, and $7481<7500$.
Iterating \eqref{eq:potential} and using
$\sum_{j=0}^{K-1}\Delta_j\le\Delta$ proves the bound on $P_j$ in
\eqref{eq:potential-expansion}. Its bound on $\sigma_{\tau_j}$
follows from $\sigma_{\tau_j}\le P_{j+1}$ and
$P_0=2\sigma_0+10^4L_1g_0\le10^4Q_0$.

To prove \eqref{eq:scale-sum-general}, sum \eqref{eq:potential}
over $j=b,\ldots,b+m-1$ and rearrange:
\[
 \frac14\sum_{j=b}^{b+m-1}P_{j+1}
 \le\frac34(P_b-P_{b+m})+4mL_0
                         +CL_1^{3/2}\sum_{j=b}^{b+m-1}\Delta_j.
\]
Now use $P_{b+m}\ge0$, $\sigma_{\tau_j}\le P_{j+1}$, and
$\sum_{j=b}^{b+m-1}\Delta_j\le\Delta$.
\end{proof}

\begin{lemma}[Initial-stage bounds]
\label{lem:initial-stage}
There exists an integer $j_0\ge0$ with
\begin{equation}\label{eq:initial-count}
 j_0=O\left(1+\lp\frac{Q_0}{Q}
              +\lp\frac{g_0}{(\Delta^2Q)^{1/3}}\right)
\end{equation}
such that
\begin{align}
 g_j&\le C(\Delta^2Q)^{1/3}
                         &&(j_0\le j\le K),\label{eq:initial-gradient}\\
 \sum_{j=b}^{b+m-1}\sigma_{\tau_j}
 &\le C\bigl(mL_0+L_1^{3/2}\Delta\bigr)
                         &&(j_0\le b<b+m\le K).
                         \label{eq:scale-sum}
\end{align}
\end{lemma}
\begin{proof}
Set
\[
 j_\sigma=\left\lceil\frac{\lp(P_0/Q)}{\log(4/3)}\right\rceil.
\]
By \eqref{eq:potential-expansion},
\begin{equation}\label{eq:scale-stabilized}
 P_j\le CQ\quad(j_\sigma\le j\le K),\qquad
 \sigma_{\tau_j}\le CQ\quad(j_\sigma\le j<K).
\end{equation}
Also $P_0\le10^4Q_0$ gives
$j_\sigma=O(1+\lp(Q_0/Q))$.
Equation~\eqref{eq:scale-sum-general} and $m\ge1$ now prove
\eqref{eq:scale-sum} for every $b\ge j_\sigma$.

For $j\in\mathcal B$,
\begin{equation}\label{eq:boundary-gradient}
 g_{j+1}\underset{(a)}{\le}51\sigma_{\tau_j}r_{\tau_j}^2
       \underset{(b)}{\le}
       \frac{51}{3^{2/3}}\sigma_{\tau_j}^{1/3}\Delta_j^{2/3}.
\end{equation}
Here (a) is \eqref{eq:boundary} and (b) uses
$3\sigma_{\tau_j}r_{\tau_j}^3\le\Delta_j$ from
\eqref{eq:descent-sum}.
For $j\in\mathcal I$, $g_{j+1}\le11g_j/16$. Since
$\Delta_j\le\Delta$ and $\sigma_{\tau_j}\le CQ_0$, these two
cases imply
\[
 g_j\le\max\{g_0,C(\Delta^2Q_0)^{1/3}\}\qquad(0\le j\le K).
\]
After $j_\sigma$, the sharper scale bound
\eqref{eq:scale-stabilized} gives
\[
 g_{j+1}\le
 \begin{cases}
 C(\Delta^2Q)^{1/3},&j\in\mathcal B,\\
 (11/16)g_j,&j\in\mathcal I.
 \end{cases}
\]
Choose a common universal $C\ge1$ for the preceding two displays.
Their iteration yields, for $j_\sigma\le j\le K$,
\[
 g_j\le\max\left\{
 (11/16)^{j-j_\sigma}
       \max\{g_0,C(\Delta^2Q_0)^{1/3}\},\
 C(\Delta^2Q)^{1/3}\right\}.
\]
It therefore suffices to take
\begin{equation}\label{eq:initial-index}
 j_0=j_\sigma+
 \left\lceil
 \frac{\displaystyle\lp\frac{g_0}{(\Delta^2Q)^{1/3}}
             +\frac13\log\frac{Q_0}{Q}}
      {\log(16/11)}
 \right\rceil.
\end{equation}
This proves \eqref{eq:initial-gradient} and \eqref{eq:initial-count}.
\end{proof}
