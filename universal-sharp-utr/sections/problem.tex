\section{Preliminaries}
\label{sec:setting}
We consider $\min_{x\in\R^n}f(x)$ under the generalized smoothness
condition of \citet[Assumption~1]{fang2026parameter}.

\begin{assumption}[Generalized smoothness]\label{ass:setting}
The function $f\in C^3(\R^n)$ is bounded below:
$f_*:=\inf_{x\in\R^n}f(x)>-\infty$. There exist constants
$L_0>0$ and $L_1\ge0$ such that
\begin{equation}\label{eq:assumption}
 \norm{\nabla^3 f(x)}\le L_0+L_1\norm{\nabla f(x)}
 \qquad\text{for every }x\in\R^n.
\end{equation}
\end{assumption}

For independent tolerances
$\epsilon,\delta>0$, an $(\epsilon,\delta)$-second-order stationary
point satisfies
\begin{equation}\label{eq:stationarity}
 \norm{\nabla f(x)}\le\epsilon,\qquad
 \lambda_{\min}(\nabla^2 f(x))\ge-\delta.
\end{equation}

\begin{lemma}\label{lem:taylor-remainders}
For any $x_k,s_k\in\R^n$, let
$M_k=\max_{0\le u\le1}\norm{\nabla^3f(x_k+us_k)}$. Then
\begin{subequations}
\begin{align}
 \bigl|f(x_k+s_k)-f(x_k)-\nabla f(x_k)^Ts_k
               -\tfrac12s_k^T\nabla^2f(x_k)s_k\bigr|
 &\le \tfrac16 M_k\norm{s_k}^3, \label{eq:taylor-value}\\
 \norm{\nabla f(x_k+s_k)-\nabla f(x_k)-\nabla^2f(x_k)s_k}
 &\le \tfrac12 M_k\norm{s_k}^2. \label{eq:taylor-gradient}
\end{align}
\end{subequations}
\end{lemma}
\begin{proof}
Apply Taylor's formula with integral remainder to $f$ and $\nabla f$ along the segment from $x_k$ to $x_k+s_k$.
\end{proof}
